6/// Internal helper to get the best available python command (local venv vs system)
7fn get_python_command() -> Command {
8    // We expect to be running from typescript/src-tauri in dev mode
9    let venv_path = "../../.venv/bin/python";
10    if Path::new(venv_path).exists() {
11        Command::new(venv_path)
12    } else {
13        Command::new("python3")
14    }
15}

17/// Lists available trained machine learning models in the local directory.
18pub fn list_trained_models() -> Result<Vec<String>, String> {
19    let path = Path::new("../../model_weights");
20    if !path.exists() {
21        return Ok(vec![]);
22    }
23
24    let mut models = Vec::new();
25    if let Ok(entries) = std::fs::read_dir(path) {
26        for entry in entries {
27            if let Ok(entry) = entry {
28                let path = entry.path();
29                if path.extension().map_or(false, |ext| ext == "pt") {
30                    if let Some(name) = path.file_stem().and_then(|s| s.to_str()) {
31                        models.push(name.to_string());
32                    }
33                }
34            }
35        }
36    }
37    models.sort();
38    Ok(models)
39}

57/// Runs inference on a selected trained model using the Python runtime.
58pub async fn predict_trained_model(
59    model_name: String,
60    input_data: Vec<f64>,
61) -> Result<Vec<f64>, String> {
62    let model_path = format!("../../model_weights/{}.pt", model_name);
63
64    let mut cmd = get_python_command();
65    cmd.arg("../../python/src/infer.py")
66        .arg("--model_path")
67        .arg(&model_path)
68        .arg("--input_json")
69        .arg(serde_json::to_string(&input_data).unwrap_or_else(|_| "[]".to_string()));
70
71    let output = cmd
72        .output()
73        .map_err(|e| format!("Failed to execute python: {}", e))?;
74
75    if !output.status.success() {
76        let stderr = String::from_utf8_lossy(&output.stderr);
77        return Err(format!("Inference script failed: {}", stderr));
78    }
79
80    let stdout = String::from_utf8_lossy(&output.stdout);
81    let result: InferenceOutput = serde_json::from_str(&stdout)
82        .map_err(|e| format!("Failed to parse inference output: {} | raw: {}", e, stdout))?;
83
84    if result.status == "success" {
85        if let Some(val) = result.prediction {
86            if let Some(arr) = val.as_array() {
87                let vec: Vec<f64> = arr.iter().filter_map(|v| v.as_f64()).collect();
88                return Ok(vec);
89            } else if let Some(num) = val.as_f64() {
90                return Ok(vec![num]);
91            }
92        }
93        Err("Empty or invalid prediction format".to_string())
94    } else {
95        Err(result
96            .message
97            .unwrap_or_else(|| "Unknown error".to_string()))
98    }
99}

163/// Trains a supervised learning model from CSV data with progress callbacks.
164pub async fn train_model_session<F>(
165    params: TrainingParams,
166    on_progress: F,
167) -> Result<String, String>
168where
169    F: Fn(TrainingProgress) + Send + Sync + 'static,
170{
171    let model_params_str =
172        serde_json::to_string(&params.model_params).unwrap_or_else(|_| "{}".to_string());
173
174    let mut cmd = get_python_command();
175    cmd.arg("-m")
176        .arg("pipeline.lightning.supervised_learning")
177        .arg("--csv_path")
178        .arg(&params.csv_path)
179        .arg("--target_column")
180        .arg(&params.target_column)
181        .arg("--model_name")
182        .arg(&params.model_name)
183        .arg("--epochs")
184        .arg(params.epochs.to_string())
185        .arg("--batch_size")
186        .arg(params.batch_size.to_string())
187        .arg("--learning_rate")
188        .arg(params.learning_rate.to_string())
189        .arg("--seq_len")
190        .arg(params.seq_len.to_string())
191        .arg("--pred_len")
192        .arg(params.pred_len.to_string())
193        .arg("--train_split")
194        .arg(params.train_split.to_string())
195        .arg("--model_params")
196        .arg(&model_params_str)
197        .current_dir("../../python/src")
198        .stdout(Stdio::piped())
199        .stderr(Stdio::piped());
200
201    let mut child = cmd
202        .spawn()
203        .map_err(|e| format!("Failed to spawn python: {}", e))?;
204
205    let stdout = child
206        .stdout
207        .take()
208        .ok_or_else(|| "Failed to get stdout".to_string())?;
209    let reader = std::io::BufReader::new(stdout);
210
211    let mut final_model_path: Option<String> = None;
212
213    // Use a blocking thread since Stdout reading is blocking
214    let result = tokio::task::spawn_blocking(move || {
215        for line in reader.lines() {
216            if let Ok(line) = line {
217                if let Ok(progress) = serde_json::from_str::<TrainingProgress>(&line) {
218                    on_progress(progress.clone());
219
220                    if progress.msg_type == "complete" {
221                        final_model_path = progress.model_path;
222                    } else if progress.msg_type == "error" {
223                        return Err(progress
224                            .message
225                            .unwrap_or_else(|| "Training error".to_string()));
226                    }
227                }
228            }
229        }
230
231        let status = child
232            .wait()
233            .map_err(|e| format!("Failed to wait for process: {}", e))?;
234        if !status.success() {
235            return Err("Training process failed".to_string());
236        }
237
238        final_model_path.ok_or_else(|| "No model path returned".to_string())
239    })
240    .await
241    .map_err(|e| format!("Task failed: {}", e))?;
242
243    result
244}
