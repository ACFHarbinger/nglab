1/*!
2 * nglab - High-Performance RL Arena for Financial Trading
3 *
4 * This crate provides a Rust-based simulation engine for reinforcement learning
5 * in financial markets, exposing a Gymnasium-compatible interface to Python via PyO3.
6 *
7 * # Core Components
8 *
9 * - **Arena**: The central simulation controller that manages the environment lifecycle,
10 *   time-stepping, and agent interactions.
11 * - **OrderBook**: A high-performance Central Limit Order Book (CLOB) implementation
12 *   supporting price-time priority, limit/market orders, and realistic fill simulation.
13 * - **TradingEnv**: An OpenAI Gym-compatible environment wrapper that handles observation
14 *   normalization, reward calculation, and action space mapping.
15 *
16 * # Architecture
17 *
18 * The system is designed as a hybrid Rust/Python application:
19 * - **Rust Layer**: Handles the computationally intensive simulation loop, order matching,
20 *   and state management. It aims for zero-allocation in the hot path.
21 * - **Python Layer**: Consumes the Rust primitives via PyO3, providing a familiar API for
22 *   ML researchers using PyTorch, TensorFlow, or JAX.
23 *
24 * # Example Usage
25 *
26 * While primarily designed to be used from Python, the core components can be used natively in Rust:
27 *
28 * ```rust,no_run
29 * use nglab::simulation::TradingEnv;
30 *
31 * // Initialize environment with starting capital and transaction costs
32 * let mut env = TradingEnv::new(10_000.0, 0.001, 10, 100, false);
33 *
34 * // Reset environment
35 * let initial_obs = env.reset_rs();
36 *
37 * // Step through simulation
38 * let action = 1; // e.g., Buy
39 * let (obs, reward, terminated, truncated, info) = env.step_rs(action);
40 *
41 * println!("Reward: {}, Portfolio: {}", reward, info.portfolio_value);
42 * ```
43 */

61/**
62 * Arena - The main simulation environment.
63 *
64 * Manages the state and stepping of the trading simulation.
65 */
66#[cfg_attr(feature = "python", pyclass)]
67pub struct Arena {
68    /** Total number of steps taken in the arena */
69    step_count: u64,
70}
71

76#[cfg(feature = "python")]

94impl Arena {

99        Arena { step_count: 0 }

101
102    /**
103     * Get the current step count of the simulation.
104     *
105     * Returns the number of steps that have been executed.

108        self.step_count

110}

118/**
119 * Python module entry point for the `nglab` extension.
120 *
121 * Exposes Rust classes and methods to the Python environment.
122 */
123#[cfg(feature = "python")]
124#[pymodule]
125fn _nglab(m: &Bound<'_, PyModule>) -> PyResult<()> {
126    m.add_class::<Arena>()?;
127    m.add_class::<simulation::orderbook::OrderBook>()?;
128    m.add_class::<simulation::polymarket::PolymarketArena>()?;
129    m.add_class::<simulation::gym::TradingEnv>()?;
130    Ok(())
131}
