11#[derive(Serialize, Deserialize, Debug, Clone, Copy, PartialEq)]
12pub enum SeasonalType {

14    Multiplicative,
15}

20/**
21 * Configuration parameters for the Holt-Winters Exponential Smoothing model.
22 */
23#[derive(Serialize, Deserialize, Debug, Clone)]
24pub struct HoltWintersParams {
25    pub alpha: f64, // Level smoothing (0-1)
26    pub beta: f64,  // Trend smoothing (0-1)
27    pub gamma: f64, // Seasonal smoothing (0-1)
28    pub period: usize,
29    pub seasonal_type: SeasonalType,
30    pub steps: usize,
31    pub sigma: f64, // Noise standard deviation
32    pub seed: Option<u64>,
33    pub data: Option<Vec<f64>>, // Historical data for initialization
34}
35
36/**
37 * Result container for Exponential Smoothing forecasts.
38 */
39#[derive(Serialize, Deserialize, Debug)]
40pub struct HoltWintersResult {
41    pub path: Vec<f64>,
42    pub used_seed: Option<u64>,
43}

45/**
46 * Run a forecast / simulation using Holt-Winters Exponential Smoothing.
47 *
48 * @param params Model parameters and optional historical data.
49 */
50pub fn simulate(params: HoltWintersParams) -> Result<HoltWintersResult, String> {
51    let seed = if let Some(s) = params.seed {
52        s
53    } else {
54        rand::rng().next_u64()
55    };
56
57    use rand::{RngCore, SeedableRng};
58    let mut rng = rand::rngs::StdRng::seed_from_u64(seed);
59
60    let m = params.period;
61    if m == 0 {
62        return Err("Period must be >= 1".to_string());
63    }
64
65    // State variables
66    // L_t: Level, T_t: Trend, S_t: Seasonals (store last m values)
67    let (mut l, mut t, mut s) = if let Some(ref data) = params.data {
68        if data.len() < 2 * m {
69            // Heuristic fallback if not enough data for proper initialization,
70            // but let's try to handle at least basic cases.
71            // If extremely short data, we might just default.
72            if data.len() < 2 {
73                return Err("Data length must be at least 2 for trend initialization".to_string());
74            }
75        }
76        initialize_from_data(&data, m, params.seasonal_type)?
77    } else {
78        // Default warm-up state if no data provided
79        let initial_l = 100.0;
80        let initial_t = 0.0;
81        let initial_s = match params.seasonal_type {
82            SeasonalType::Additive => vec![0.0; m],
83            SeasonalType::Multiplicative => vec![1.0; m],
84        };
85        (initial_l, initial_t, initial_s)
86    };
87
88    // If data is provided, we might want to "run" the filter through the data to get the final state
89    // before simulating 'steps'.
90    // If no data, we just start generating.
91
92    // BUT, the standard usage in this project (like arima/garch) seems to be:
93    // 1. If params.data is present, fit/warmup on it, then project 'steps' ahead.
94    // 2. The output 'path' usually contains ONLY the future 'steps' points?
95    //    Let's check arima.rs:
96    //    "let result_path = integrated_x.into_iter().skip(n - params.steps).collect();" -> Yes, it returns the accumulated path.
97    //    Wait, arima.rs `if let Some(ref data)` block (lines 93+) seems to append predictions to the data?
98    //    Let's look closely at `arima.rs` Logic:
99    //    It returns `current_predictions` which accumulates `next_preds`.
100    //    Wait, `ArimaResult.path` is just the predictions?
101    //    Line 94: `let mut current_predictions = result_x[start_idx..].to_vec();`
102    //    It seems it calculates the 'diff' predictions then integrates them back.
103    //    Effectively, it returns the *continuation* of the series.
104
105    // So for ES:
106    // We update state (L, T, S) iterating through `params.data` (if any).
107    // Then we generate `steps` future values, updating state stochastically if sigma > 0.
108
109    if let Some(ref data) = params.data {
110        // Run filter over history to update L, T, S
111        for i in 0..data.len() {
112            let y = data[i];
113            // let s_idx = i % m; // Unused
114
115            // We need S_{t-m}. In our circular buffer or vec logic, we need to be careful.
116            // Let's store S as a Vec of length m, representing indices [t-m, ..., t-1] relative to current t?
117            // Actually, standard HW formulation:
118            // S_t computed at time t is used for time t+m.
119            // When at time t, we use S_{t-m} (which was computed m steps ago).
120            // Let's keep `s` as a Vec<f64> of size m. `s[i % m]` will store the seasonal component for season `i`.
121            // Wait, this is tricky for updates.
122            // Easier: Just keep track of the *last known* seasonal index.
123
124            // Standard approach:
125            // S_new = gamma * (Y - L_new) + (1-gamma) * S_old  (Additive)
126            // But we need the S from "last cycle".
127            // Let's interpret `s` as: `s[k]` is the Seasonal factor for the k-th period of the cycle (0..m-1).
128            // At step i, the season index is `idx = i % m`.
129            // We use `s[idx]` as the seasonal component coming from the past.
130            // Then we update `s[idx]` with the new value.
131
132            let idx = i % m;
133            let old_s = s[idx];
134            let (new_l, new_t, new_s) = match params.seasonal_type {
135                SeasonalType::Additive => {
136                    let l_val = params.alpha * (y - old_s) + (1.0 - params.alpha) * (l + t);
137                    let t_val = params.beta * (l_val - l) + (1.0 - params.beta) * t;
138                    let s_val = params.gamma * (y - l_val) + (1.0 - params.gamma) * old_s;
139                    (l_val, t_val, s_val)
140                }
141                SeasonalType::Multiplicative => {
142                    // Avoid division by zero
143                    let safe_old_s = if old_s.abs() < 1e-9 { 1.0 } else { old_s };
144                    let l_val = params.alpha * (y / safe_old_s) + (1.0 - params.alpha) * (l + t);
145                    let t_val = params.beta * (l_val - l) + (1.0 - params.beta) * t;
146                    let s_val = params.gamma * (y / if l_val.abs() < 1e-9 { 1.0 } else { l_val })
147                        + (1.0 - params.gamma) * safe_old_s;
148                    (l_val, t_val, s_val)
149                }
150            };
151            l = new_l;
152            t = new_t;
153            s[idx] = new_s;
154        }
155    } else {
156        // If no data, we are already at "end" of initialization (t=0 effectively).
157        // But we might want some warmup steps if sigma > 0 to let randomness diverge?
158        // For now, assume parameters start from specific L, T, S.
159    }
160
161    // Forecasting / Simulation
162    let mut predictions = Vec::with_capacity(params.steps);
163    let start_idx = params.data.as_ref().map(|d| d.len()).unwrap_or(0);
164
165    // We continue updating the state step by step, adding noise.
166    // Or do we just project? "Simulation" usually implies adding noise to the recursive equations.
167    // Standard HW forecasting projects the expected value.
168    // "Simulating" paths implies: Y_{t+h} = L_{t+h-1} + T_{t+h-1} + S_{t+h-m} + \epsilon
169    // Then UPADTE L, T, S using this simulated Y.
170
171    for i in 0..params.steps {
172        let current_t = start_idx + i;
173        let idx = current_t % m;
174        let old_s = s[idx];
175
176        // 1. Generate Prediction / Value for this step
177        let noise: f64 = StandardNormal.sample(&mut rng);
178        let epsilon = noise * params.sigma;
179
180        let y_sim = match params.seasonal_type {
181            SeasonalType::Additive => l + t + old_s + epsilon,
182            SeasonalType::Multiplicative => (l + t) * old_s + epsilon, // Noise usually additive? Or multiplicative?
183                                                                       // Typically additive noise for the observation equation.
184        };
185
186        predictions.push(y_sim);
187
188        // 2. Update states (Feedback Loop)
189        // If we simply "forecast", we fix states. But "simulation" implies evolving the process with noise.
190        // We feed y_sim back into the equations.
191        let (new_l, new_t, new_s) = match params.seasonal_type {
192            SeasonalType::Additive => {
193                let l_val = params.alpha * (y_sim - old_s) + (1.0 - params.alpha) * (l + t);
194                let t_val = params.beta * (l_val - l) + (1.0 - params.beta) * t;
195                let s_val = params.gamma * (y_sim - l_val) + (1.0 - params.gamma) * old_s;
196                (l_val, t_val, s_val)
197            }
198            SeasonalType::Multiplicative => {
199                let safe_old_s = if old_s.abs() < 1e-9 { 1.0 } else { old_s };
200                let l_val = params.alpha * (y_sim / safe_old_s) + (1.0 - params.alpha) * (l + t);
201                let t_val = params.beta * (l_val - l) + (1.0 - params.beta) * t;
202                let s_val = params.gamma * (y_sim / if l_val.abs() < 1e-9 { 1.0 } else { l_val })
203                    + (1.0 - params.gamma) * safe_old_s;
204                (l_val, t_val, s_val)
205            }
206        };
207
208        l = new_l;
209        t = new_t;
210        s[idx] = new_s;
211    }
212
213    Ok(HoltWintersResult {
214        path: predictions,
215        used_seed: Some(seed),
216    })
217}

219// Simple heuristic initialization
220fn initialize_from_data(
221    data: &[f64],
222    m: usize,
223    stype: SeasonalType,
224) -> Result<(f64, f64, Vec<f64>), String> {
225    if data.len() < m {
226        // Fallback for very short data (should verify earlier)
227        return match stype {
228            SeasonalType::Additive => Ok((axis_mean(data), 0.0, vec![0.0; m])),
229            SeasonalType::Multiplicative => Ok((axis_mean(data), 0.0, vec![1.0; m])),
230        };
231    }
232
233    // Initial Level = Average of first season
234    let mut initial_l = 0.0;
235    for i in 0..m {
236        initial_l += data[i];
237    }
238    initial_l /= m as f64;
239
240    // Initial Trend
241    // Simple heuristic: (Average of last season - Average of first season) / (N - m) roughly?
242    // Or just (data[m] - data[0]) / m if we have at least m+1 points?
243    // Let's use standard Hyndman-Khandakar or similar initialization if possible, but keep it simple.
244    // Let's try: Trend = (Mean of second season - Mean of first season) / m
245    // Only if we have 2*m data points.
246    let initial_t = if data.len() >= 2 * m {
247        let mut sum_first = 0.0;
248        let mut sum_second = 0.0;
249        for i in 0..m {
250            sum_first += data[i];
251            sum_second += data[i + m];
252        }
253        (sum_second - sum_first) / (m as f64 * m as f64)
254    } else {
255        0.0 // Insufficient data for reliable trend
256    };
257
258    // Initial Seasonals
259    let mut initial_s = Vec::with_capacity(m);
260    for i in 0..m {
261        match stype {
262            SeasonalType::Additive => {
263                // S_i = y_i - L
264                // Better: S_i = y_i - (L_0 + T_0 * (i+1)) ?
265                // Simply difference from mean of first season for now.
266                initial_s.push(data[i] - initial_l);
267            }
268            SeasonalType::Multiplicative => {
269                if initial_l.abs() < 1e-9 {
270                    initial_s.push(1.0);
271                } else {
272                    initial_s.push(data[i] / initial_l);
273                }
274            }
275        }
276    }
277
278    // Normalize seasonals
279    // Additive: sum(S) should be 0.
280    // Multiplicative: sum(S) should be m.
281    match stype {
282        SeasonalType::Additive => {
283            let sum_s: f64 = initial_s.iter().sum();
284            let correction = sum_s / m as f64;
285            for val in initial_s.iter_mut() {
286                *val -= correction;
287            }
288        }
289        SeasonalType::Multiplicative => {
290            let sum_s: f64 = initial_s.iter().sum();
291            if sum_s.abs() > 1e-9 {
292                let correction = m as f64 / sum_s;
293                for val in initial_s.iter_mut() {
294                    *val *= correction;
295                }
296            }
297        }
298    }
299
300    Ok((initial_l, initial_t, initial_s))
301}
302
303fn axis_mean(data: &[f64]) -> f64 {
304    if data.is_empty() {
305        return 0.0;
306    }
307    data.iter().sum::<f64>() / data.len() as f64
308}

314    #[test]
315    fn test_hw_determinism() {
316        let params = HoltWintersParams {
317            alpha: 0.5,
318            beta: 0.1,
319            gamma: 0.1,
320            period: 4,
321            seasonal_type: SeasonalType::Additive,
322            steps: 10,
323            sigma: 1.0,
324            seed: Some(42),
325            data: Some(vec![10.0, 11.0, 12.0, 13.0, 14.0, 15.0, 16.0, 17.0]),
326        };
327
328        let res1 = simulate(params.clone()).unwrap();
329        let res2 = simulate(params.clone()).unwrap();
330
331        assert_eq!(res1.path, res2.path);
332    }

334    #[test]
335    fn test_hw_multiplicative_basic() {
336        // Multiplicative seasonality often used for exponentially growing series or variable amplitude
337        let params = HoltWintersParams {
338            alpha: 0.5,
339            beta: 0.1,
340            gamma: 0.1,
341            period: 4,
342            seasonal_type: SeasonalType::Multiplicative,
