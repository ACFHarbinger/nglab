12/**

15#[derive(Debug, Clone, Serialize, Deserialize)]
16pub struct ProphetParams {
17    pub growth: String,                   // "linear" or "flat"
18    pub changepoints: Option<Vec<usize>>, // Indices of changepoints (optional)
19    pub seasonality_mode: String,         // "additive" or "multiplicative"
20    pub yearly_seasonality: bool,
21    pub weekly_seasonality: bool,
22    pub daily_seasonality: bool,
23    pub seasonality_prior_scale: f64,
24    pub changepoint_prior_scale: f64,
25    pub forecast_horizon: usize,
26    pub times: Vec<i64>,
27    pub values: Vec<f64>,
28}

41/**
42 * A simplified Prophet implementation for time-series forecasting.
43 */
44pub struct Prophet {
45    params: ProphetParams,
46    // Model coefficients
47    k: f64, // Base trend rate
48    m: f64, // Base trend offset
49    #[allow(dead_code)]
50    deltas: Array1<f64>, // Trend rate adjustments at changepoints
51    beta: Array1<f64>, // Seasonality coefficients
52}
53
54impl Prophet {
55    pub fn new(params: ProphetParams) -> Self {
56        Self {
57            params,
58            k: 0.0,
59            m: 0.0,
60            deltas: Array1::zeros(0),
61            beta: Array1::zeros(0),
62        }
63    }
64
65    /**
66     * Fit the model to historical data using ridge regression.
67     *
68     * @param times Array of timestamps.
69     * @param y Array of target values.
70     */
71    pub fn fit(&mut self, times: &[i64], y: &[f64]) -> Result<(), String> {
72        if times.len() != y.len() || times.is_empty() {
73            return Err("Data mismatch or empty".to_string());
74        }
75
76        let t = self.normalize_time(times);
77        let y_vec = Array1::from_vec(y.to_vec());
78        let n = t.len();
79
80        // 1. Setup Trend Component
81        // For simple linear trend: y = (k + A(t)delta) * t + (m + A(t)(-t_changes)delta)
82        // Check params for changepoints. If none, just simple linear: y = kt + m
83        // Implemented: Simple Linear Trend (Global) for now to start "Prophet-Lite"
84        // TODO: Add changepoints support
85
86        // 2. Setup Seasonality Component (Fourier Series)
87        let seasonal_features = self.make_seasonality_features(times);
88        let num_seasonal_params = seasonal_features.ncols();
89
90        // Design Matrix X: [t, 1, seasonal_features...]
91        // Note: For simple linear trend, we just need column `t` and column `1` (bias).
92
93        let mut x_mat = Array2::<f64>::zeros((n, 2 + num_seasonal_params));
94
95        for i in 0..n {
96            x_mat[[i, 0]] = t[i]; // Trend feature
97            x_mat[[i, 1]] = 1.0; // Bias/Offset feature
98
99            for j in 0..num_seasonal_params {
100                x_mat[[i, 2 + j]] = seasonal_features[[i, j]];
101            }
102        }
103
104        // Ridge Regression: (X^T X + lambda*I)^-1 X^T y
105        // lambda (regularization) depends on priors.
106        // For simplicity in this Lite version, we use a small fixed lambda for stability
107        // or derive from params.
108        let lambda = 0.01;
109        let coeffs = self.solve_ridge(&x_mat, &y_vec, lambda)?;
110
111        // Unpack coefficients
112        self.k = coeffs[0];
113        self.m = coeffs[1];
114
115        if num_seasonal_params > 0 {
116            self.beta = coeffs.slice(ndarray::s![2..]).to_owned();
117        } else {
118            self.beta = Array1::zeros(0);
119        }
120
121        Ok(())
122    }

124    // --- Helpers ---
125
126    fn normalize_time(&self, times: &[i64]) -> Vec<f64> {
127        if times.is_empty() {
128            return vec![];
129        }
130        let min_t = times[0] as f64;
131        let max_t = times[times.len() - 1] as f64;
132        let range = (max_t - min_t).max(1.0);
133
134        times.iter().map(|&t| (t as f64 - min_t) / range).collect()
135    }
136
137    fn make_seasonality_features(&self, times: &[i64]) -> Array2<f64> {
138        let n = times.len();
139        let mut features = Vec::new();
140
141        // Unix timestamp in seconds expected
142        // Daily (period = 1 day = 86400s) - order 4
143        if self.params.daily_seasonality {
144            let period = 86400.0;
145            self.add_fourier_terms(times, period, 4, &mut features);
146        }
147
148        // Weekly (period = 7 days) - order 3
149        if self.params.weekly_seasonality {
150            let period = 86400.0 * 7.0;
151            self.add_fourier_terms(times, period, 3, &mut features);
152        }
153
154        // Yearly (period = 365.25 days) - order 10
155        if self.params.yearly_seasonality {
156            let period = 86400.0 * 365.25;
157            self.add_fourier_terms(times, period, 10, &mut features);
158        }
159
160        if features.is_empty() {
161            return Array2::zeros((n, 0));
162        }
163
164        let num_features = features.len();
165        let mut mat = Array2::zeros((n, num_features));
166        for (j, col) in features.iter().enumerate() {
167            for i in 0..n {
168                mat[[i, j]] = col[i];
169            }
170        }
171        mat
172    }
173
174    fn add_fourier_terms(
175        &self,
176        times: &[i64],
177        period: f64,
178        order: usize,
179        output: &mut Vec<Vec<f64>>,
180    ) {
181        for i in 1..=order {
182            let f = i as f64;
183            // sin(2 * pi * i * t / P)
184            // cos(2 * pi * i * t / P)
185            // Note: t should be in seconds if period is in seconds.
186
187            let sin_term: Vec<f64> = times
188                .iter()
189                .map(|&t| (2.0 * PI * f * t as f64 / period).sin())
190                .collect();
191            let cos_term: Vec<f64> = times
192                .iter()
193                .map(|&t| (2.0 * PI * f * t as f64 / period).cos())
194                .collect();
195
196            output.push(sin_term);
197            output.push(cos_term);
198        }
199    }
200
201    /** Solves (X^T X + lambda I) beta = X^T y using Cholesky Decomposition */
202    fn solve_ridge(
203        &self,
204        x: &Array2<f64>,
205        y: &Array1<f64>,
206        lambda: f64,
207    ) -> Result<Array1<f64>, String> {
208        let xt = x.t();
209        let xt_x = xt.dot(x);
210        let xt_y = xt.dot(y);
211
212        let n_dims = xt_x.nrows();
213        let mut a = xt_x;
214
215        // Add ridge regularization
216        for i in 0..n_dims {
217            a[[i, i]] += lambda;
218        }
219
220        // Cholesky: A = L L^T
221        // Solve Ly = b (forward subst)
222        // Solve L^T x = y (backward subst)
223
224        let l = self.cholesky(&a)?;
225        let z = self.forward_substitution(&l, &xt_y)?;
226        let beta = self.backward_substitution(&l.t().to_owned(), &z)?;
227
228        Ok(beta)
229    }
230
231    fn cholesky(&self, a: &Array2<f64>) -> Result<Array2<f64>, String> {
232        let n = a.nrows();
233        let mut l = Array2::zeros((n, n));
234
235        for i in 0..n {
236            for j in 0..=i {
237                let mut sum = 0.0;
238                for k in 0..j {
239                    sum += l[[i, k]] * l[[j, k]];
240                }
241
242                if i == j {
243                    let val = a[[i, i]] - sum;
244                    if val <= 0.0 {
245                        return Err("Matrix not positive definite".to_string());
246                    }
247                    l[[i, j]] = val.sqrt();
248                } else {
249                    l[[i, j]] = (a[[i, j]] - sum) / l[[j, j]];
250                }
251            }
252        }
253        Ok(l)
254    }
255
256    fn forward_substitution(
257        &self,
258        l: &Array2<f64>,
259        b: &Array1<f64>,
260    ) -> Result<Array1<f64>, String> {
261        let n = l.nrows();
262        let mut y = Array1::zeros(n);
263
264        for i in 0..n {
265            let mut sum = 0.0;
266            for j in 0..i {
267                sum += l[[i, j]] * y[j];
268            }
269            if l[[i, i]].abs() < 1e-10 {
270                return Err("Singular matrix in forward substitution".to_string());
271            }
272            y[i] = (b[i] - sum) / l[[i, i]];
273        }
274        Ok(y)
275    }
276
277    fn backward_substitution(
278        &self,
279        u: &Array2<f64>,
280        y: &Array1<f64>,
281    ) -> Result<Array1<f64>, String> {
282        let n = u.nrows();
283        let mut x = Array1::zeros(n);
284
285        for i in (0..n).rev() {
286            let mut sum = 0.0;
287            for j in i + 1..n {
288                sum += u[[i, j]] * x[j];
289            }
290            if u[[i, i]].abs() < 1e-10 {
291                return Err("Singular matrix in backward substitution".to_string());
292            }
293            x[i] = (y[i] - sum) / u[[i, i]];
294        }
295        Ok(x)
296    }
297}

299/**
300 * Run a forecast / simulation using the Prophet model.
301 *
302 * @param params Prophet configuration and historical data.
303 */
304pub fn simulate(params: ProphetParams) -> Result<ProphetResult, String> {
305    let historical_times = &params.times;
306    let historical_values = &params.values;
307    // 1. Fit Model
308    let mut model = Prophet::new(params.clone());
309    model.fit(historical_times, historical_values)?;
310
311    // 2. Predict Future
312    // Generate future timestamps
313    let last_time = *historical_times.last().unwrap_or(&0);
314    // Determine step size (median step from history or just daily/hourly?)
315    // Basic heuristic: average diff
316    let step_size = if historical_times.len() > 1 {
317        let diffs: Vec<i64> = historical_times.windows(2).map(|w| w[1] - w[0]).collect();
318        diffs.iter().sum::<i64>() / diffs.len() as i64
319    } else {
320        86400 // Default 1 day
321    };
322
323    let horizon = params.forecast_horizon;
324    let mut future_times = Vec::with_capacity(horizon);
325    let mut current_time = last_time;
326    for _ in 0..horizon {
327        current_time += step_size;
328        future_times.push(current_time);
329    }
330
331    // Prepare features for prediction
332    // Need to use the SAME scalar logic as fit!
333    // Since we created the model cleanly, we should refactor 'fit' to store 'min_t'/'scale_t' in struct
334    // Re-instantiating `t_norm` based on the fitted range.
335
336    let t_min = historical_times[0] as f64;
337    let t_max = historical_times[historical_times.len() - 1] as f64;
338    let t_scale = (t_max - t_min).max(1.0);
339
340    let future_t_norm: Vec<f64> = future_times
341        .iter()
342        .map(|&t| (t as f64 - t_min) / t_scale)
343        .collect();
344
345    let future_seasonal_features = model.make_seasonality_features(&future_times); // Uses raw times for Fourier
346
347    let mut predicted_values = Vec::with_capacity(horizon);
348    let mut predicted_trend = Vec::with_capacity(horizon);
349    let mut predicted_seasonal = Vec::with_capacity(horizon);
350
351    // Calculate predictions
352    // y = (k * t + m) + (seasonality)
353    let has_seasonality = future_seasonal_features.ncols() > 0;
354
355    for i in 0..horizon {
356        let t = future_t_norm[i];
357        let trend = model.k * t + model.m;
358
359        let mut seasonal = 0.0;
360        if has_seasonality {
361            for j in 0..model.beta.len() {
362                seasonal += model.beta[j] * future_seasonal_features[[i, j]];
363            }
364        }
365
366        let y_hat = if params.seasonality_mode == "multiplicative" {
367            trend * (1.0 + seasonal)
368        } else {
369            trend + seasonal
370        };
371
372        predicted_values.push(y_hat);
373        predicted_trend.push(trend);
374        predicted_seasonal.push(seasonal);
375    }
376
377    Ok(ProphetResult {
378        times: future_times,
379        values: predicted_values,
380        trend: predicted_trend,
381        seasonal: predicted_seasonal,
382    })
383}

389    #[test]
390    fn test_prophet_linear_trend() {
391        // Synthetic data: y = 2*t + 5
392        let n = 100;
393        let times: Vec<i64> = (0..n).map(|i| i * 86400).collect();
394        let values: Vec<f64> = times
395            .iter()
396            .enumerate()
397            .map(|(i, _)| 2.0 * i as f64 + 5.0)
398            .collect();
399
400        let params = ProphetParams {
401            growth: "linear".to_string(),
402            changepoints: None,
403            seasonality_mode: "additive".to_string(),
404            yearly_seasonality: false,
405            weekly_seasonality: false,
406            daily_seasonality: false,
407            seasonality_prior_scale: 10.0,
408            changepoint_prior_scale: 0.05,
409            forecast_horizon: 10,
410            times: times.clone(),
411            values: values.clone(),
412        };
413
414        let result = simulate(params).unwrap();
415
416        assert_eq!(result.values.len(), 10);
417        // Next value should be approx 2 * 100 + 5 = 205
418        // Trend is fitted on normalized time 0..1.
419        // t=100 corresponds to normalized t > 1.
420        // Correct check: The continuity of the line.
421        let first_pred = result.values[0];
422        assert!(
423            (first_pred - 205.0).abs() < 1.0,
424            "Prediction {} should be close to 205.0",
425            first_pred
426        );
427    }
