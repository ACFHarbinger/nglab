23/**
24 * Action space types for the agent.
25 */
26#[derive(Debug, Clone, Copy)]
27pub enum ActionType {
28    /** Hold current position */
29    Hold,
30    /** Buy (long) */
31    Buy,
32    /** Sell (short) */
33    Sell,
34}
35
36impl From<i32> for ActionType {
37    fn from(value: i32) -> Self {
38        match value {
39            0 => ActionType::Hold,
40            1 => ActionType::Buy,
41            2 => ActionType::Sell,
42            _ => ActionType::Hold,
43        }
44    }
45}

47/**
48 * Step result returned to high-level callers (e.g., Python).
49 *
50 * Bundles the new observation, the reward received for the action,
51 * and flags indicating if the episode has finished.
52 *
53 * # Fields
54 *
55 * * `observation` - Vector of feature values for the new state.
56 * * `reward` - Scaled reward value (e.g., risk-adjusted return).
57 * * `terminated` - True if the agent reached a terminal state (e.g., bankruptcy).
58 * * `truncated` - True if the simulation was cut short (e.g., time limit).
59 * * `info` - Additional diagnostic metadata.
60 */
61#[derive(Debug, Clone, serde::Serialize, serde::Deserialize)]
62pub struct StepResult {
63    pub observation: Vec<f64>,
64    pub reward: f64,
65    pub terminated: bool,
66    pub truncated: bool,
67    pub info: StepInfo,
68}

82/**
83 * Observation buffer for high-performance zero-copy access.
84 */
85pub struct ObservationBuffer {
86    data: Vec<f64>,
87    shape: (usize, usize),
88}
89
90impl ObservationBuffer {
91    pub fn new(features: usize, lookback: usize) -> Self {
92        ObservationBuffer {
93            data: vec![0.0; features * lookback],
94            shape: (lookback, features),
95        }
96    }
97
98    pub fn update(&mut self, row: usize, values: &[f64]) {
99        let start = row * self.shape.1;
100        let end = start + values.len().min(self.shape.1);
101        self.data[start..end].copy_from_slice(&values[..end - start]);
102    }
103}

105/**
106 * Trading environment with Gymnasium interface.
107 *
108 * Wraps market simulation logic into a standard RL format suitable for training agents.
109 * It manages the interaction between the agent (action) and the market (order book),
110 * computing rewards and updating state variables.
111 *
112 * # Features
113 *
114 * - **Zero-Copy Observations**: Leverages `numpy` crate for efficient data transfer to Python.
115 * - **Configurable Rewards**: Supports various reward schemes (PnL, Sharpe Ratio, etc.).
116 * - **Market Simulation**: Integrated with `OrderBook` for realistic execution.
117 *
118 * # Fields
119 *
120 * * `orderbook` - The central limit order book managing market liquidity.
121 * * `position` - Current net position (positive for long, negative for short).
122 * * `cash` - Available cash balance for trading.
123 * * `portfolio_value` - Total account equity (Cash + Mark-to-Market Position).
124 * * `step_count` - Current time step in the episode.
125 */
126#[cfg_attr(feature = "python", pyclass)]
127pub struct TradingEnv {
128    /** Order book for simulation */
129    orderbook: OrderBook,
130    /** Price history for backtesting */
131    prices: Vec<f64>,
132    /** Current step index */
133    current_step: usize,
134    /** Current position (positive = long, negative = short) */
135    position: f64,
136    /** Current cash balance */
137    cash: f64,
138    /** Initial capital */
139    initial_capital: f64,
140    /** Transaction cost rate */
141    transaction_cost: f64,
142    /** Lookback window for observations */
143    lookback: usize,
144    /** Number of features per timestep */
145    num_features: usize,
146    /** Returns history for Sharpe calculation */
147    returns: Vec<f64>,
148    /** Previous portfolio value for return calculation */
149    prev_portfolio_value: f64,
150    /** Maximum steps before truncation */
151    max_steps: usize,
152    /** Total steps taken */
153    total_steps: u64,
154    /** Rerun logger for visualization */
155    logger: crate::utils::visualizer::RerunLogger,
156}

189    /** Reset the environment */
190    #[pyo3(signature = (seed=None, options=None))]
191    #[instrument(skip(self, py), fields(step = self.total_steps))]
192    pub fn reset<'py>(
193        &mut self,
194        py: Python<'py>,
195        seed: Option<u64>,
196        options: Option<PyObject>,
197    ) -> PyResult<(Bound<'py, PyArray2<f64>>, PyObject)> {
198        if let Some(s) = seed {
199            info!("Reseting environment with seed: {}", s);
200            // TODO: Seed RNG
201        } else {
202            info!("Reseting environment with random seed");
203        }
204        self.reset_rs();
205
206        let obs = self.get_observation(py)?;
207        let info = PyDict::new(py).into();
208        Ok((obs, info))
209    }
210
211    /** Take a step in the environment */
212    #[instrument(skip(self, py), fields(step = self.total_steps, action = action))]
213    pub fn step<'py>(
214        &mut self,
215        py: Python<'py>,
216        action: i32,
217    ) -> PyResult<(Bound<'py, PyArray2<f64>>, f64, bool, bool, PyObject)> {
218        let action_type = ActionType::from(action);
219        let lookback = self.lookback;
220        let num_features = self.num_features;
221
222        debug!("Executing action: {:?}", action_type);
223
224        let trade_cost = self.execute_action(action_type);
225        self.current_step += 1;
226        self.total_steps += 1;
227
228        let portfolio_value = self.portfolio_value();
229        let returns = (portfolio_value - self.prev_portfolio_value) / self.prev_portfolio_value;
230        self.returns.push(returns);
231        self.prev_portfolio_value = portfolio_value;
232
233        let reward = self.calculate_reward(returns, trade_cost);
234        let terminated =
235            portfolio_value <= 0.0 || self.current_step >= self.prices.len().saturating_sub(1);
236        let truncated = self.current_step.saturating_sub(self.lookback) >= self.max_steps;
237
238        let obs_data = self.generate_observation_data();
239
240        let step_info = StepInfo {
241            portfolio_value,
242            position: self.position,
243            cash: self.cash,
244            sharpe_ratio: self.calculate_sharpe(30),
245            total_steps: self.total_steps,
246        };
247
248        self.logger
249            .log_step(self.total_steps, &self.orderbook, portfolio_value);
250
251        // Build info dict
252        let dict = PyDict::new(py);
253        dict.set_item("portfolio_value", step_info.portfolio_value)?;
254        dict.set_item("position", step_info.position)?;
255        dict.set_item("cash", step_info.cash)?;
256        dict.set_item("sharpe_ratio", step_info.sharpe_ratio)?;
257        dict.set_item("total_steps", step_info.total_steps)?;
258        let info = dict.into();
259
260        // Create numpy array
261        let obs_array = ndarray::Array2::from_shape_vec((lookback, num_features), obs_data)
262            .map_err(|e| ArenaError::DataLoading(format!("Invalid observation shape: {}", e)))?;
263        let obs = obs_array.to_pyarray(py);
264
265        Ok((obs, reward, terminated, truncated, info))
266    }
267
268    /** Get current observation as numpy array */
269    pub fn get_observation<'py>(&self, py: Python<'py>) -> PyResult<Bound<'py, PyArray2<f64>>> {
270        let obs_data = self.generate_observation_data();
271
272        // Reshape to (lookback, features)
273        let array = ndarray::Array2::from_shape_vec((self.lookback, self.num_features), obs_data)
274            .map_err(|e| ArenaError::DataLoading(format!("Invalid shape: {}", e)))?;
275
276        Ok(array.to_pyarray(py))
277    }
278}

384    /** Pure Rust step function (no Python dependency) */
385    #[instrument(skip(self), fields(step = self.total_steps))]
386    pub fn step_rs(&mut self, action: i32) -> (Vec<f64>, f64, bool, bool, StepInfo) {
387        let action_type = ActionType::from(action);
388
389        // Execute action
390        let trade_cost = self.execute_action(action_type);
391
392        // Advance time
393        self.current_step += 1;
394        self.total_steps += 1;
395
396        // Calculate reward
397        let portfolio_value = self.portfolio_value();
398        let returns = (portfolio_value - self.prev_portfolio_value) / self.prev_portfolio_value;
399        self.returns.push(returns);
400        self.prev_portfolio_value = portfolio_value;
401
402        // Risk-adjusted reward (Sharpe-like)
403        let reward = self.calculate_reward(returns, trade_cost);
404
405        // Check termination
406        let terminated =
407            portfolio_value <= 0.0 || self.current_step >= self.prices.len().saturating_sub(1);
408        let truncated = self.current_step.saturating_sub(self.lookback) >= self.max_steps;
409
410        // Generate observation data
411        let obs_data = self.generate_observation_data();
412
413        // Collect info data
414        let step_info = StepInfo {
415            portfolio_value,
416            position: self.position,
417            cash: self.cash,
418            sharpe_ratio: self.calculate_sharpe(30),
419            total_steps: self.total_steps,
420        };
421
422        // Log simulation state (thread-safe)
423        self.logger
424            .log_step(self.total_steps, &self.orderbook, portfolio_value);
425
426        (obs_data, reward, terminated, truncated, step_info)
427    }

429    /** Internal method to generate observation data (pure Rust, no GIL) */
430    fn generate_observation_data(&self) -> Vec<f64> {
431        let mut obs = vec![0.0f64; self.lookback * self.num_features];
432
433        for i in 0..self.lookback {
434            let idx = self
435                .current_step
436                .saturating_sub(self.lookback)
437                .saturating_add(i);
438            if idx < self.prices.len() {
439                let price = self.prices[idx];
440                let prev_price = if idx > 0 { self.prices[idx - 1] } else { price };
441                let returns = if prev_price > 0.0 {
442                    (price - prev_price) / prev_price
443                } else {
444                    0.0
445                };
446
447                let row_start = i * self.num_features;
448                obs[row_start] = price / self.prices[0]; // Normalized price?
449                obs[row_start + 1] = returns; // Returns
450                obs[row_start + 2] = 0.0; // Volume (placeholder)
451                obs[row_start + 3] = self.orderbook.imbalance(); // Order book imbalance
452                obs[row_start + 4] = self.position / self.initial_capital; // Normalized position
453                obs[row_start + 5] = self.cash / self.initial_capital; // Normalized cash
454            }
455        }
456        obs
457    }

464    /** Execute a trading action */
465    fn execute_action(&mut self, action: ActionType) -> f64 {
466        let price = match self.current_price() {
467            Some(p) => p,
468            None => return 0.0,
469        };
470
471        let trade_size = self.initial_capital * 0.1; // Fixed position sizing
472        let mut cost = 0.0;
473
474        match action {
475            ActionType::Hold => {}
476            ActionType::Buy => {
477                let shares = trade_size / price;
478                let tx_cost = trade_size * self.transaction_cost;
479
480                if self.cash >= trade_size + tx_cost {
481                    self.cash -= trade_size + tx_cost;
482                    self.position += shares;
483                    cost = tx_cost;
484
485                    // Update orderbook with our trade
486                    self.orderbook.submit_market_order(shares, Side::Bid);
487                }
488            }
489            ActionType::Sell => {
490                if self.position > 0.0 {
491                    let shares_to_sell = (trade_size / price).min(self.position);
492                    let proceeds = shares_to_sell * price;
493                    let tx_cost = proceeds * self.transaction_cost;
494
495                    self.position -= shares_to_sell;
496                    self.cash += proceeds - tx_cost;
497                    cost = tx_cost;
498
499                    self.orderbook
500                        .submit_market_order(shares_to_sell, Side::Ask);
501                }
502            }
503        }
504
505        cost
506    }

508    /** Calculate risk-adjusted reward */
509    fn calculate_reward(&self, returns: f64, trade_cost: f64) -> f64 {
510        // Simple reward: returns minus transaction costs
511        let base_reward = returns * 100.0; // Scale up small returns
512
513        // Penalty for holding costs (encourage action when appropriate)
514        let cost_penalty = trade_cost / self.initial_capital * 100.0;
515
516        // Drawdown penalty
517        let max_value: f64 = self
518            .returns
519            .iter()
520            .scan(self.initial_capital, |acc, &r| {
521                *acc *= 1.0 + r;
522                Some(*acc)
523            })
524            .fold(self.initial_capital, f64::max);
525
526        let current_value = self.portfolio_value();
527        let drawdown = if max_value > 0.0 {
528            (max_value - current_value) / max_value
529        } else {
530            0.0
531        };
532        let drawdown_penalty = drawdown * 0.1;
533
534        base_reward - cost_penalty - drawdown_penalty
535    }
536
537    /** Calculate rolling Sharpe ratio */
538    fn calculate_sharpe(&self, window: usize) -> f64 {
539        if self.returns.len() < 2 {
540            return 0.0;
541        }
542
543        let recent: Vec<_> = self.returns.iter().rev().take(window).copied().collect();
544
545        if recent.len() < 2 {
546            return 0.0;
547        }
548
549        let mean: f64 = recent.iter().sum::<f64>() / recent.len() as f64;
550        let variance: f64 =
551            recent.iter().map(|r| (r - mean).powi(2)).sum::<f64>() / (recent.len() - 1) as f64;
552        let std = variance.sqrt();
553
554        if std > 0.0 {
555            mean / std * (252.0_f64).sqrt() // Annualized
556        } else {
557            0.0
558        }
559    }
