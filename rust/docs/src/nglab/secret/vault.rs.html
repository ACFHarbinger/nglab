14/// Summary of a vault entry (for listing)

31/// Standard response wrapper for vault operations
32#[derive(Debug, Serialize, Deserialize)]
33pub struct VaultResponse<T> {
34    pub success: bool,
35    pub message: String,
36    pub data: Option<T>,
37}
38
39impl<T> VaultResponse<T> {
40    pub fn success(message: &str, data: Option<T>) -> Self {
41        Self {
42            success: true,
43            message: message.to_string(),
44            data,
45        }
46    }
47
48    pub fn error(message: &str) -> Self {
49        Self {
50            success: false,
51            message: message.to_string(),
52            data: None,
53        }
54    }
55}

88    /// Open an encrypted connection to the database
89    pub fn open_connection(&self, key: &str) -> SqlResult<Connection> {
90        let conn = Connection::open(&self.db_path)?;
91        // SQLCipher encryption key
92        conn.pragma_update(None, "key", &key)?;
93        Ok(conn)
94    }
95
96    /// Initialize the database schema
97    pub fn init_db(&self, key: &str) -> SqlResult<()> {
98        let conn = self.open_connection(key)?;
99        conn.execute(
100            "CREATE TABLE IF NOT EXISTS secrets (
101                id INTEGER PRIMARY KEY AUTOINCREMENT,
102                label TEXT NOT NULL,
103                value TEXT NOT NULL,
104                created_at DATETIME DEFAULT CURRENT_TIMESTAMP
105            )",
106            [],
107        )?;
108        Ok(())
109    }
110
111    /// Derive an encryption key string from a master password using Argon2
112    pub fn derive_key_string(master_password: &str, salt: &[u8]) -> String {
113        let mut key = [0u8; 32];
114        let argon2 = Argon2::default();
115        let salt_str = SaltString::encode_b64(salt).expect("Failed to encode salt");
116
117        let _ = argon2.hash_password_into(
118            master_password.as_bytes(),
119            salt_str.as_str().as_bytes(),
120            &mut key,
121        );
122
123        // Return as hex-encoded string for SQLCipher
124        key.iter().map(|b| format!("{:02x}", b)).collect()
125    }

127    /// Add a secret to the vault
128    pub fn add_secret(&self, key_str: &str, label: &str, value: &str) -> SqlResult<i64> {
129        let conn = self.open_connection(key_str)?;
130        conn.execute(
131            "INSERT INTO secrets (label, value) VALUES (?1, ?2)",
132            params![label, value],
133        )?;
134        Ok(conn.last_insert_rowid())
135    }

137    /// List all secrets (summaries only)
138    pub fn list_secrets(&self, key_str: &str) -> SqlResult<Vec<VaultSummary>> {
139        let conn = self.open_connection(key_str)?;
140        let mut stmt = conn.prepare("SELECT id, label, created_at FROM secrets")?;
141        let rows = stmt.query_map([], |row| {
142            Ok(VaultSummary {
143                id: row.get(0)?,
144                label: row.get(1)?,
145                created_at: row.get(2)?,
146            })
147        })?;
148
149        let mut entries = Vec::new();
150        for row in rows {
151            entries.push(row?);
152        }
153        Ok(entries)
154    }
155
156    /// Get a specific secret
157    pub fn get_secret(&self, key_str: &str, id: i64) -> SqlResult<Option<VaultEntry>> {
158        let conn = self.open_connection(key_str)?;
159        let mut stmt =
160            conn.prepare("SELECT id, label, value, created_at FROM secrets WHERE id = ?1")?;
161        let mut rows = stmt.query_map(params![id], |row| {
162            Ok(VaultEntry {
163                id: row.get(0)?,
164                label: row.get(1)?,
165                value: row.get(2)?,
166                created_at: row.get(3)?,
167            })
168        })?;
169
170        if let Some(entry_res) = rows.next() {
171            return Ok(Some(entry_res?));
172        }
173        Ok(None)
174    }
175
176    /// Delete a secret
177    pub fn delete_secret(&self, key_str: &str, id: i64) -> SqlResult<()> {
178        let conn = self.open_connection(key_str)?;
179        conn.execute("DELETE FROM secrets WHERE id = ?1", params![id])?;
180        Ok(())
181    }
